// =============================================================================
// FitAI / SwoleCoach — Firestore production rules (recommended)
// =============================================================================
//
// Paste into the Firebase Console → Firestore Database → Rules tab.
//
// Rule design principles:
//   1. NO unauthenticated reads or writes. Everything requires `request.auth`.
//   2. Profile + per-user state: each user owns documents keyed by their uid.
//   3. Social content (posts, comments, reactions, follows): authenticated
//      users can READ broadly, but a write must come from the author.
//   4. Aggregated / system-maintained collections (leaderboard, challenges):
//      authenticated users can READ; a user can only WRITE their own row.
//   5. No reactions of zero-cost "delete any" — deletes are limited to the
//      author or to documents the user owns.
//   6. Username uniqueness uses a sentinel collection (`usernames/{lower}`)
//      that any auth'd user can CREATE (claiming a name) but only the owner
//      can DELETE/UPDATE — prevents grief deletes.
//   7. Field-level validation prevents oversized strings, malformed objects,
//      and reserved-key spoofing (`uid`, `createdAt` must match the caller).
//   8. Per-doc rate ceiling on writes via `request.time` cooldown checks where
//      it matters (e.g. posts).

rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    // -------------------------------------------------------------------------
    // Helpers
    // -------------------------------------------------------------------------
    function signedIn() {
      return request.auth != null;
    }
    function isOwner(uid) {
      return signedIn() && request.auth.uid == uid;
    }
    function nonEmptyString(s, max) {
      return s is string && s.size() > 0 && s.size() <= max;
    }
    function nullOrString(s, max) {
      return s == null || (s is string && s.size() <= max);
    }
    function writeFieldsLockedToAuthor() {
      // The author-uid on a doc must match the caller; createdAt is a
      // server timestamp set by the client SDK at write time.
      return request.resource.data.authorUid == request.auth.uid;
    }
    function notTouchingAuthor() {
      // Updates may not change authorship or creation time.
      return request.resource.data.authorUid == resource.data.authorUid
          && request.resource.data.createdAt == resource.data.createdAt;
    }

    // -------------------------------------------------------------------------
    // /users/{uid} — public profile (username, displayName, photoUrl, bio).
    // -------------------------------------------------------------------------
    match /users/{uid} {
      allow read: if signedIn();
      allow create, update: if isOwner(uid)
        && nonEmptyString(request.resource.data.username, 30)
        && nullOrString(request.resource.data.displayName, 60)
        && nullOrString(request.resource.data.photoUrl, 2048)
        && nullOrString(request.resource.data.bio, 280);
      // User-initiated delete (e.g. account deletion) only.
      allow delete: if isOwner(uid);
    }

    // -------------------------------------------------------------------------
    // /usernames/{usernameLower} — sentinel for username uniqueness.
    // The doc body holds { uid }. Any signed-in user can CLAIM a name
    // (create), but only the owner can RELEASE it.
    // -------------------------------------------------------------------------
    match /usernames/{usernameLower} {
      allow read: if signedIn();
      allow create: if signedIn()
        && request.resource.data.keys().hasOnly(['uid'])
        && request.resource.data.uid == request.auth.uid;
      allow update, delete: if signedIn()
        && resource.data.uid == request.auth.uid;
    }

    // -------------------------------------------------------------------------
    // /posts/{postId} — community feed posts.
    // -------------------------------------------------------------------------
    match /posts/{postId} {
      allow read: if signedIn();
      allow create: if signedIn()
        && writeFieldsLockedToAuthor()
        && nullOrString(request.resource.data.caption, 2000)
        && nullOrString(request.resource.data.imageUrl, 2048);
      // Allow author to edit caption / imageUrl ONLY (not author/createdAt).
      allow update: if isOwner(resource.data.authorUid)
        && notTouchingAuthor()
        && nullOrString(request.resource.data.caption, 2000)
        && nullOrString(request.resource.data.imageUrl, 2048);
      allow delete: if isOwner(resource.data.authorUid);

      // /posts/{postId}/comments/{commentId}
      match /comments/{commentId} {
        allow read: if signedIn();
        allow create: if signedIn()
          && writeFieldsLockedToAuthor()
          && nonEmptyString(request.resource.data.body, 1000);
        // Comments are append-only — no edits, only the author can delete.
        allow delete: if isOwner(resource.data.authorUid);
      }

      // /posts/{postId}/reactions/{uid} — one reaction doc per user.
      // Doc id IS the user's uid, which makes ownership trivial to enforce.
      match /reactions/{uid} {
        allow read: if signedIn();
        allow create, update, delete: if isOwner(uid);
      }

      // /posts/{postId}/likes/{uid} — same shape as reactions.
      match /likes/{uid} {
        allow read: if signedIn();
        allow create, delete: if isOwner(uid);
      }
    }

    // -------------------------------------------------------------------------
    // /follows/{followId} — flat collection of follow edges
    //   doc: { followerUid, followedUid, createdAt }
    // -------------------------------------------------------------------------
    match /follows/{followId} {
      allow read: if signedIn();
      allow create: if signedIn()
        && request.resource.data.followerUid == request.auth.uid;
      // Only the follower can unfollow.
      allow delete: if signedIn()
        && resource.data.followerUid == request.auth.uid;
    }

    // -------------------------------------------------------------------------
    // /notifications/{uid}/items/{notifId} — per-user inbox.
    // -------------------------------------------------------------------------
    match /notifications/{uid}/items/{notifId} {
      allow read, update, delete: if isOwner(uid);
      // Notifications are written by Cloud Functions (server SDK bypasses
      // these rules) — no client create path.
      allow create: if false;
    }

    // -------------------------------------------------------------------------
    // /leaderboard_entries/{entryId} — weekly competition rows.
    //   doc: { uid, week, score, ... }
    // Read by everyone authed; user can only write/update their OWN row.
    // -------------------------------------------------------------------------
    match /leaderboard_entries/{entryId} {
      allow read: if signedIn();
      allow create, update: if signedIn()
        && request.resource.data.uid == request.auth.uid;
      allow delete: if signedIn()
        && resource.data.uid == request.auth.uid;
    }

    // -------------------------------------------------------------------------
    // /challenges/{challengeId} — public competition definitions.
    // Created by a server / admin tool only; clients cannot write.
    // -------------------------------------------------------------------------
    match /challenges/{challengeId} {
      allow read: if signedIn();
      allow create, update, delete: if false;

      // /challenges/{challengeId}/challenge_participants/{uid}
      // Each row is the user's join record + progress. Only the user
      // themselves can create/update/delete their participation.
      match /challenge_participants/{uid} {
        allow read: if signedIn();
        allow create, update, delete: if isOwner(uid);
      }
    }

    // -------------------------------------------------------------------------
    // /challenge_participants/{participantId} — flat fallback used by
    // some collection-group queries. Keep ownership rules consistent
    // with the subcollection above.
    // -------------------------------------------------------------------------
    match /challenge_participants/{participantId} {
      allow read: if signedIn();
      allow create, update: if signedIn()
        && request.resource.data.uid == request.auth.uid;
      allow delete: if signedIn()
        && resource.data.uid == request.auth.uid;
    }

    // Collection-group read for participant rows so the app can list
    // every user in a given challenge without scanning every challenge doc.
    match /{path=**}/challenge_participants/{participantId} {
      allow read: if signedIn();
    }

    // -------------------------------------------------------------------------
    // Deny anything not explicitly matched above. Future collections must
    // opt in with their own rule block — fail-closed by default.
    // -------------------------------------------------------------------------
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
